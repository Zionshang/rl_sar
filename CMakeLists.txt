cmake_minimum_required(VERSION 3.18)
project(rl_sar VERSION 4.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BUILD_MUJOCO "Build the Go2/Go2W MuJoCo simulator" ON)
option(BUILD_REAL "Build optional Go2/Go2W hardware deployment (Linux)" OFF)
option(ENABLE_TORCH "Enable TorchScript policies" ON)
option(ENABLE_ONNX "Enable ONNX policies" ON)
option(ENABLE_JOYSTICK "Enable Linux USB gamepad support" OFF)
option(ENABLE_CSV_LOGGER "Record motor data to CSV" OFF)
include(CTest)
set(POLICY_DIR "${PROJECT_SOURCE_DIR}/policy" CACHE PATH "Policy/configuration directory")
set(ASSET_DIR "${PROJECT_SOURCE_DIR}/assets" CACHE PATH "Go2/Go2W MuJoCo asset directory")
set(MUJOCO_ROOT "${PROJECT_SOURCE_DIR}/library/mujoco" CACHE PATH "MuJoCo installation")
set(LIBTORCH_ROOT "${PROJECT_SOURCE_DIR}/library/inference_runtime/libtorch" CACHE PATH "LibTorch installation")
set(ONNXRUNTIME_ROOT "${PROJECT_SOURCE_DIR}/library/inference_runtime/onnxruntime" CACHE PATH "ONNX Runtime installation")
set(UNITREE_SDK2_DIR "${PROJECT_SOURCE_DIR}/library/unitree_sdk2" CACHE PATH "Optional Unitree SDK2 source directory")
set(JOYSTICK_DIR "${PROJECT_SOURCE_DIR}/library/joystick" CACHE PATH "Optional joystick source directory")
if((BUILD_MUJOCO OR BUILD_REAL) AND NOT (ENABLE_TORCH OR ENABLE_ONNX))
    message(FATAL_ERROR "Enable at least one policy backend: ENABLE_TORCH or ENABLE_ONNX")
endif()
find_package(Threads REQUIRED)
find_package(TBB REQUIRED)
find_package(yaml-cpp REQUIRED)
if(TARGET yaml-cpp::yaml-cpp)
    set(YAML_CPP_TARGET yaml-cpp::yaml-cpp)
else()
    set(YAML_CPP_TARGET yaml-cpp)
endif()
add_library(inference_runtime core/inference_runtime/inference_runtime.cpp)
target_include_directories(inference_runtime PUBLIC core/inference_runtime core/logger)
if(ENABLE_TORCH)
    list(PREPEND CMAKE_PREFIX_PATH "${LIBTORCH_ROOT}")
    find_package(Torch REQUIRED)
    target_compile_definitions(inference_runtime PUBLIC USE_TORCH)
    target_link_libraries(inference_runtime PUBLIC ${TORCH_LIBRARIES})
    separate_arguments(TORCH_OPTIONS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
    target_compile_options(inference_runtime PUBLIC ${TORCH_OPTIONS})
endif()
if(ENABLE_ONNX)
    find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h
        HINTS "${ONNXRUNTIME_ROOT}/include" PATH_SUFFIXES onnxruntime/core/session)
    find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib")
    if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIBRARY)
        message(FATAL_ERROR "ONNX Runtime not found. Run ./setup_deps.sh onnx, set ONNXRUNTIME_ROOT, or use -DENABLE_ONNX=OFF")
    endif()
    add_library(onnxruntime_prebuilt SHARED IMPORTED)
    set_target_properties(onnxruntime_prebuilt PROPERTIES
        IMPORTED_LOCATION "${ONNXRUNTIME_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${ONNXRUNTIME_INCLUDE_DIR}")
    target_compile_definitions(inference_runtime PUBLIC USE_ONNX)
    target_link_libraries(inference_runtime PUBLIC onnxruntime_prebuilt)
endif()
add_library(observation_buffer core/observation_buffer/observation_buffer.cpp)
target_include_directories(observation_buffer PUBLIC core/observation_buffer)
add_library(rl_core core/rl_sdk/rl_sdk.cpp)
target_include_directories(rl_core PUBLIC
    core/rl_sdk core/fsm core/loop core/vector_math core/logger fsm_robot)
target_compile_definitions(rl_core PUBLIC POLICY_DIR="${POLICY_DIR}")
target_link_libraries(rl_core PUBLIC inference_runtime observation_buffer TBB::tbb Threads::Threads ${YAML_CPP_TARGET})
if(ENABLE_CSV_LOGGER)
    target_compile_definitions(rl_core PUBLIC CSV_LOGGER)
endif()
if(BUILD_MUJOCO)
    # The bundled Simulate UI is from MuJoCo 3.2.7: use matching headers/library.
    if(APPLE AND EXISTS "${MUJOCO_ROOT}/Headers/mujoco.h")
        set(MUJOCO_COMPAT_INCLUDE_DIR "${CMAKE_BINARY_DIR}/mujoco_include")
        file(MAKE_DIRECTORY "${MUJOCO_COMPAT_INCLUDE_DIR}/mujoco")
        file(GLOB MUJOCO_HEADERS "${MUJOCO_ROOT}/Headers/*.h")
        file(COPY ${MUJOCO_HEADERS} DESTINATION "${MUJOCO_COMPAT_INCLUDE_DIR}/mujoco")
        set(MUJOCO_INCLUDE_DIR "${MUJOCO_COMPAT_INCLUDE_DIR}")
    else()
        find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h HINTS "${MUJOCO_ROOT}/include")
    endif()
    find_library(MUJOCO_LIBRARY NAMES mujoco mujoco.3.2.7 libmujoco.so.3.2.7
        HINTS "${MUJOCO_ROOT}/lib" "${MUJOCO_ROOT}/Versions/Current")
    if(NOT MUJOCO_INCLUDE_DIR OR NOT MUJOCO_LIBRARY)
        message(FATAL_ERROR "MuJoCo 3.2.7 not found. Run ./setup_deps.sh mujoco or set MUJOCO_ROOT")
    endif()
    file(STRINGS "${MUJOCO_INCLUDE_DIR}/mujoco/mujoco.h" MUJOCO_HEADER_VERSION REGEX "^#define mjVERSION_HEADER ")
    if(NOT MUJOCO_HEADER_VERSION MATCHES "mjVERSION_HEADER +327($| )")
        message(FATAL_ERROR "The bundled MuJoCo UI requires 3.2.7; set MUJOCO_ROOT to a matching installation")
    endif()
    find_package(glfw3 REQUIRED)
    add_library(mujoco_prebuilt SHARED IMPORTED)
    set_target_properties(mujoco_prebuilt PROPERTIES
        IMPORTED_LOCATION "${MUJOCO_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${MUJOCO_INCLUDE_DIR}")
    file(GLOB MUJOCO_UI_SOURCES CONFIGURE_DEPENDS third_party/mujoco_simulate/*.cc)
    if(APPLE)
        enable_language(OBJCXX)
        file(GLOB MUJOCO_UI_OBJCXX CONFIGURE_DEPENDS third_party/mujoco_simulate/*.mm)
        list(APPEND MUJOCO_UI_SOURCES ${MUJOCO_UI_OBJCXX})
    endif()
    add_library(mujoco_ui ${MUJOCO_UI_SOURCES})
    target_include_directories(mujoco_ui PUBLIC third_party/mujoco_simulate)
    target_link_libraries(mujoco_ui PUBLIC mujoco_prebuilt glfw Threads::Threads ${CMAKE_DL_LIBS})
    if(APPLE)
        target_link_libraries(mujoco_ui PUBLIC "-framework CoreVideo" "-framework OpenGL" "-framework AppKit")
    endif()
    add_executable(rl_sim_mujoco src/rl_sim_mujoco.cpp)
    target_include_directories(rl_sim_mujoco PRIVATE include)
    target_compile_definitions(rl_sim_mujoco PRIVATE ASSET_DIR="${ASSET_DIR}")
    target_link_libraries(rl_sim_mujoco PRIVATE rl_core mujoco_ui)
    if(ENABLE_JOYSTICK)
        if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
            message(FATAL_ERROR "USB gamepad support requires Linux")
        endif()
        if(NOT EXISTS "${JOYSTICK_DIR}/joystick.cc")
            message(FATAL_ERROR "Joystick source not found. Run ./setup_deps.sh joystick")
        endif()
        target_sources(rl_sim_mujoco PRIVATE "${JOYSTICK_DIR}/joystick.cc")
        target_include_directories(rl_sim_mujoco PRIVATE "${JOYSTICK_DIR}")
        target_compile_definitions(rl_sim_mujoco PRIVATE ENABLE_JOYSTICK)
    endif()
endif()
if(BUILD_REAL)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "Go2/Go2W hardware deployment requires Linux")
    endif()
    if(NOT EXISTS "${UNITREE_SDK2_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "Unitree SDK2 not found. Run ./setup_deps.sh real or set UNITREE_SDK2_DIR")
    endif()
    set(BUILD_EXAMPLES OFF CACHE BOOL "Disable Unitree SDK examples" FORCE)
    add_subdirectory("${UNITREE_SDK2_DIR}" "${CMAKE_BINARY_DIR}/unitree_sdk2")
    add_executable(rl_real_go2 src/rl_real_go2.cpp)
    target_include_directories(rl_real_go2 PRIVATE include)
    target_link_libraries(rl_real_go2 PRIVATE rl_core unitree_sdk2)
endif()
if(BUILD_TESTING)
    add_executable(test_observation_buffer tests/test_observation_buffer.cpp)
    target_link_libraries(test_observation_buffer PRIVATE observation_buffer)
    add_test(NAME observation_buffer COMMAND test_observation_buffer)
    add_executable(test_vector_math tests/test_vector_math.cpp)
    target_include_directories(test_vector_math PRIVATE core/vector_math)
    add_test(NAME vector_math COMMAND test_vector_math)
    add_executable(test_policy_contract tests/test_policy_contract.cpp)
    target_link_libraries(test_policy_contract PRIVATE rl_core)
    add_test(NAME policy_contract COMMAND test_policy_contract)
    add_executable(test_policy_inference tests/test_policy_inference.cpp)
    target_link_libraries(test_policy_inference PRIVATE rl_core)
    target_compile_definitions(test_policy_inference PRIVATE TEST_FIXTURE_DIR="${PROJECT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME policy_inference COMMAND test_policy_inference)
    if(BUILD_MUJOCO)
        add_executable(test_mujoco_models tests/test_mujoco_models.cpp)
        target_include_directories(test_mujoco_models PRIVATE include)
        target_link_libraries(test_mujoco_models PRIVATE rl_core mujoco_prebuilt)
        target_compile_definitions(test_mujoco_models PRIVATE ASSET_DIR="${ASSET_DIR}")
        add_test(NAME mujoco_models COMMAND test_mujoco_models)
    endif()
endif()
message(STATUS "Go2/Go2W: MuJoCo=${BUILD_MUJOCO}, real=${BUILD_REAL}, Torch=${ENABLE_TORCH}, ONNX=${ENABLE_ONNX}, joystick=${ENABLE_JOYSTICK}")
